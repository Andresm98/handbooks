No necesitas diseñar Kafka para necesitar entender sistemas distribuidos. Cada vez que tu código hace una llamada HTTP a un servicio externo, lee de una base de datos remota, o escribe en una cola — estás operando en un sistema distribuido con todos sus problemas: fallos de red, latencia variable, mensajes duplicados, y estados inconsistentes.

Idempotencia: El Principio que Salva Datos


  Por qué los retries ocurren aunque el servidor respondió exitosamente

  Cliente                    Red                    Servidor
     │                        │                         │
     │──POST /payments──────►│──────────────────────►  │
     │                        │                         │ Procesa pago ✓
     │                        │◄─── Timeout / Drop ──   │ Envía 200 OK
     │                        │  (respuesta perdida)    │
     │   Timeout!             │                         │
     │──POST /payments──────►│──────────────────────►  │
     │  (retry)               │                         │ Procesa pago OTRA VEZ ❌
     │◄──────────────────────│◄────────────────────────│ 201 Created

  El cliente nunca recibió el 200 original.
  El servidor procesó el pago DOS veces.
  El cliente tiene razón al reintentar.
  El servidor tiene razón al procesar.
  Resultado: cobro duplicado.

  Solución: Idempotency Key
  El cliente genera un UUID antes del primer intento.
  Lo envía en cada retry con el MISMO UUID.
  El servidor: "¿Ya vi este UUID? → Retornar el resultado anterior."

// Implementación de idempotency keys en servidor
async function processPayment(
  req: Request,
  paymentData: PaymentDTO
): Promise<PaymentResult> {

  const idempotencyKey = req.headers['idempotency-key']
  if (!idempotencyKey) {
    throw new BadRequestError('Idempotency-Key header required')
  }

  // Verificar si ya procesamos esta operación
  const cached = await redis.get(`idempotency:payment:${idempotencyKey}`)
  if (cached) {
    // Retornar el resultado original — sin re-procesar
    return JSON.parse(cached) as PaymentResult
  }

  // Primer intento: procesar y guardar resultado
  const result = await paymentProvider.charge(paymentData)

  // Guardar por 24 horas (ventana de retries del cliente)
  await redis.setex(
    `idempotency:payment:${idempotencyKey}`,
    86400,
    JSON.stringify(result)
  )

  return result
}

// Cliente: generar la key ANTES del primer request
async function chargeWithRetry(paymentData: PaymentDTO): Promise<PaymentResult> {
  const idempotencyKey = crypto.randomUUID()  // Generado una vez

  for (let attempt = 0; attempt < 3; attempt++) {
    try {
      return await fetch('/payments', {
        method: 'POST',
        headers: {
          'Idempotency-Key': idempotencyKey,  // Misma key en cada retry
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(paymentData)
      })
    } catch (error) {
      if (attempt === 2) throw error
      await sleep(Math.pow(2, attempt) * 1000)  // Exponential backoff
    }
  }
}

Retry con Backoff Exponencial y Circuit Breaker

# Retry con exponential backoff + jitter — el estándar de producción
import asyncio
import random
from typing import TypeVar, Callable

T = TypeVar('T')

async def retry_with_backoff(
    func: Callable[[], T],
    max_attempts: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 30.0,
    retryable_exceptions: tuple = (ConnectionError, TimeoutError)
) -> T:
    """
    Exponential backoff con jitter:
    intento 1: espera 1s + random(0-0.5s)
    intento 2: espera 2s + random(0-1s)
    intento 3: espera 4s + random(0-2s)

    El jitter evita el "thundering herd": si 1000 clientes fallan
    al mismo tiempo y todos esperan exactamente 2s, el servidor
    recibe 1000 requests simultáneos al segundo 2 → otro fallo.
    Con jitter: los 1000 se distribuyen en una ventana de tiempo.
    """
    last_exception = None

    for attempt in range(max_attempts):
        try:
            return await func()
        except retryable_exceptions as e:
            last_exception = e
            if attempt == max_attempts - 1:
                break
            delay = min(base_delay * (2 ** attempt), max_delay)
            jitter = random.uniform(0, delay * 0.5)
            await asyncio.sleep(delay + jitter)

    raise last_exception

# Circuit Breaker: evitar llamadas a servicios que están caídos

from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"       # Normal: deja pasar las llamadas
    OPEN = "open"           # Fallo: rechaza inmediatamente
    HALF_OPEN = "half_open" # Probando: deja pasar una llamada

class CircuitBreaker:
    def __init__(self, failure_threshold=5, recovery_timeout=60):
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.last_failure_time = None

    async def call(self, func):
        if self.state == CircuitState.OPEN:
            if self._should_attempt_recovery():
                self.state = CircuitState.HALF_OPEN
            else:
                raise CircuitOpenError("Circuit is OPEN — service unavailable")
        try:
            result = await func()
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise

    def _on_success(self):
        self.failure_count = 0
        self.state = CircuitState.CLOSED

    def _on_failure(self):
        self.failure_count += 1
        self.last_failure_time = time.time()
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
Conexión técnica

Los patrones de sistemas distribuidos del día a día implementan principios arquitectónicos fundamentales: la idempotencia hace posible el at-least-once delivery seguro. El circuit breaker es el patrón que implementa la resiliencia y tolerancia a fallos a nivel de código. El exponential backoff con jitter evita el thundering herd que puede convertir un fallo temporal en un fallo total en cascada.
